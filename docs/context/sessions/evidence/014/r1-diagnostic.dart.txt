import 'dart:io';
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:seguimiento/ui/exercise_figure_3d.dart';
import 'package:seguimiento/ui/exercise_figure_3d_view.dart';
import '../test/support/test_fonts.dart';
void main(){testWidgets('R1 gap source and actual lateral capture',(tester) async {
  await loadTestFonts(tester);
  final cat=Figure3DCatalog.fromJsonString(File(Figure3DCatalog.asset).readAsStringSync());
  final fig=cat.forExercise('Retracciones escapulares')!;
  var ankleGap=100.0,soleGap=100.0;
  for(var i=0;i<=60;i++) {final body=liftSkeleton(fig.skeletonAt(i/60),fig);
    for(final sd in ['near','far']) {final ankle=body.p['${sd}Ankle']!,toe=body.p['${sd}Toe']!;
      final forward=(toe-ankle).unit,lateral=body.up.cross(body.front).unit;
      final width=(lateral-forward*forward.dot(lateral)).unit;
      final sole=width.cross(forward).unit;
      final points=<V3>[ankle-forward*1.3+sole*1.1,ankle-forward*1.0-sole*.6,
        ankle+forward*.8-sole*.75,ankle+forward*(fig.dims.foot-.6)-sole*.05,
        ankle+forward*fig.dims.foot+sole*.55,ankle+forward*(fig.dims.foot-.7)+sole*1.15,
        ankle-forward*.9+sole*1.25];
      if(-ankle.y<ankleGap)ankleGap=-ankle.y;
      for(final p in points) {if(-p.y<soleGap)soleGap=-p.y;}
    }
  }
  // ignore: avoid_print
  print('R1 Retracciones61samples min ankle-floor=$ankleGap u; exact rendered shoe sole-floor=$soleGap u (~${soleGap*3.2}cm)');
  final key=GlobalKey();
  await tester.pumpWidget(MaterialApp(home:Center(child:RepaintBoundary(key:key,child:SizedBox(width:360,height:410,
    child:ColoredBox(color:const Color(0xff0b0806),child:CustomPaint(painter:Figure3DPainter(fig,0,yaw:0))))))));
  await tester.pump();
  await tester.runAsync(() async {final boundary=key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
    final image=await boundary.toImage(pixelRatio:3);final bytes=await image.toByteData(format:ui.ImageByteFormat.png);
    File('build/mockups/calentamiento/r1-retracciones-lateral-before.png').writeAsBytesSync(bytes!.buffer.asUint8List());image.dispose();});
  expect(soleGap,greaterThan(0));
});}
