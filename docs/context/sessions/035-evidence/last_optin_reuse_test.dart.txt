import 'dart:convert';
import 'dart:io';
import 'dart:ui' as ui;
import 'package:flutter_test/flutter_test.dart';
import 'package:seguimiento/ui/exercise_figure_3d.dart' as current;
import 'reuse_last_baseline.dart' as baseline;

void main() {
  testWidgets('last opt in changes preserve all other sheet and movement pixels', (tester) async {
    final old=baseline.Figure3DCatalog.fromJsonString(File('build/qa/035/reuse_last_baseline.json').readAsStringSync());
    final now=current.Figure3DCatalog.fromJsonString(File(current.Figure3DCatalog.asset).readAsStringSync());
    final entries=(jsonDecode(File('build/qa/035/reuse-last-inventory.json').readAsStringSync()) as List).cast<Map<String,Object?>>();
    var compared=0;
    final failures=<String>[];
    await tester.runAsync(() async {
      for(final entry in entries) {
        final name=entry['name'] as String, key=entry['physicalVariantKey'] as String?;
        final width=entry['width'] as int, height=entry['height'] as int;
        final viewport=ui.Size(width.toDouble(),height.toDouble());
        final a=old.forExercise(name,physicalVariantKey:key)!, b=now.forExercise(name,physicalVariantKey:key)!;
        for(final sample in (entry['samples'] as List).cast<Map<String,Object?>>()) {
          final progress=(sample['progress'] as num).toDouble();
          final yaw=(sample['yaw'] as num).toDouble(), pitch=(sample['pitch'] as num).toDouble();
          final buffers=<List<int>>[];
          for(var version=0;version<2;version++) {
            final recorder=ui.PictureRecorder();final canvas=ui.Canvas(recorder);
            if(version==0) { baseline.paintFigure3D(canvas,viewport,a,progress,yaw:yaw,pitch:pitch); }
            else { current.paintFigure3D(canvas,viewport,b,progress,yaw:yaw,pitch:pitch); }
            final picture=recorder.endRecording();final image=await picture.toImage(width,height);
            buffers.add((await image.toByteData(format:ui.ImageByteFormat.rawRgba))!.buffer.asUint8List());
            image.dispose();picture.dispose();
          }
          var equal=true;
          for(var i=0;i<buffers[0].length;i++) { if(buffers[0][i]!=buffers[1][i]) {equal=false;break;} }
          if(!equal) failures.add('$name/$key/$progress/$yaw');
          compared++;
        }
      }
    });
    File('docs/context/sessions/035-evidence/last-optin-reuse-pixel-result.json').writeAsStringSync(jsonEncode({
      'baselineSource':'1b9bc194fcbae7978b7c8fb653572fe88ba1610e',
      'candidateSource':Platform.environment['ILLUSTRATION_035_SOURCE_SHA'],
      'sheetSamples':402,'clipFigureSamples':110,'compared':compared,'failures':failures,
      'viewports':[[360,315],[320,285]],'method':'rawRgba exact comparison of real production painters; no tolerance'}));
    expect(compared,512);expect(failures,isEmpty);
  });
}
