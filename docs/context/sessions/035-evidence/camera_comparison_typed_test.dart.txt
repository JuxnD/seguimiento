import 'dart:convert';
import 'dart:io';
import 'dart:ui' as ui;
import 'package:flutter_test/flutter_test.dart';
import 'package:seguimiento/ui/exercise_figure_3d.dart' as current;
import '../../../test/support/test_fonts.dart';
import 'exercise_figure_3d_pre_camera.dart' as previous;

void main() {
  testWidgets('camera scope and legacy pull-up pixels remain explicit', (tester) async {
    await loadTestFonts(tester);
    final source = File(current.Figure3DCatalog.asset).readAsStringSync();
    final now = current.Figure3DCatalog.fromJsonString(source);
    final old = previous.Figure3DCatalog.fromJsonString(source);
    final names = ((jsonDecode(source) as Map<String, Object?>)['figures'] as List)
        .cast<Map<String, Object?>>()
        .map((f) => f['name'] as String);
    final rows = <Map<String, Object?>>[];
    for (final name in names) {
      final a = now.forExercise(name)!;
      final b = old.forExercise(name)!;
      final ac = current.cameraFor(a, const ui.Size(320, 285), yaw: a.initialYaw, pitch: a.initialPitch);
      final bc = previous.cameraFor(b, const ui.Size(320, 285), yaw: b.initialYaw, pitch: b.initialPitch);
      final currentValues = [ac.center.x, ac.center.y, ac.center.z, ac.scale];
      final oldValues = [bc.center.x, bc.center.y, bc.center.z, bc.scale];
      if (!a.hasMotion) expect(currentValues, oldValues, reason: name);
      rows.add({'name': name, 'authored': a.hasMotion, 'old': oldValues, 'current': currentValues});
    }
    final a = now.forExercise('Dominadas')!;
    final b = old.forExercise('Dominadas')!;
    final elbowProjection = <Map<String, Object?>>[];
    for (final yaw in [0.0, a.initialYaw, 35.0, 90.0]) {
      final camera = current.cameraFor(a, const ui.Size(320, 285), yaw: yaw, pitch: a.initialPitch);
      for (final progress in [.35, .45, .5, .55, .65, .75]) {
        final body = current.liftSkeleton(a.skeletonAt(progress), a);
        final near = camera.project(body.p['nearElbow']!);
        final far = camera.project(body.p['farElbow']!);
        final diameter = 1.15 * (near.k + far.k); // Actual elbow capsule radii.
        elbowProjection.add({'yaw': yaw, 'progress': progress,
          'separationPx': (near.offset - far.offset).distance,
          'combinedRadiusPx': diameter,
          'separationRatio': (near.offset - far.offset).distance / diameter});
      }
    }
    final manifest = jsonDecode(File('build/qa/035/motion-batch2/manifest-5.json').readAsStringSync()) as List;
    final samples = ((manifest.single as Map<String, Object?>)['samples'] as List).cast<Map<String, Object?>>();
    double sampleProgress(Map<String, Object?> sample) =>
        (((sample['figures'] as List).first as Map<String, Object?>)['progress'] as num).toDouble();
    for (final sample in samples) {
      final progress = sampleProgress(sample);
      final ap = current.liftSkeleton(a.skeletonAt(progress), a).p;
      final bp = previous.liftSkeleton(b.skeletonAt(progress), b).p;
      for (final key in ap.keys) {
        expect([ap[key]!.x, ap[key]!.y, ap[key]!.z], [bp[key]!.x, bp[key]!.y, bp[key]!.z]);
      }
    }
    var pixelPairs = 0;
    await tester.runAsync(() async {
      for (var index = 0; index < samples.length; index += 23) {
        final progress = sampleProgress(samples[index]);
        final bytes = <List<int>>[];
        for (final oldRenderer in [false, true]) {
          final recorder = ui.PictureRecorder();
          final canvas = ui.Canvas(recorder)..drawColor(const ui.Color(0xff0b0806), ui.BlendMode.src);
          if (oldRenderer) {
            previous.paintFigure3D(canvas, const ui.Size(320, 285), b, progress, yaw: 35, pitch: b.initialPitch);
          } else {
            current.paintFigure3D(canvas, const ui.Size(320, 285), a, progress, yaw: 35, pitch: a.initialPitch);
          }
          final picture = recorder.endRecording();
          final image = await picture.toImage(320, 285);
          bytes.add((await image.toByteData(format: ui.ImageByteFormat.rawRgba))!.buffer.asUint8List());
          image.dispose();
          picture.dispose();
        }
        expect(bytes[0], bytes[1], reason: 'pull-up frame $index');
        pixelPairs++;
      }
    });
    File('docs/context/sessions/035-evidence/camera-comparison.json').writeAsStringSync(
      const JsonEncoder.withIndent('  ').convert({'oldSource': 'eef6e067b4a3487247be3d5523acb15bb4af930a',
        'currentProduct': '8a8c94fc496a498f393e0ddeae8cfe756923c085',
        'cameraRows': rows, 'pullupWorldSamples': samples.length, 'pullupPixelPairs': pixelPairs,
        'pullupInitialYaw': a.initialYaw, 'elbowProjection': elbowProjection}));
  });
}
